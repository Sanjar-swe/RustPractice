4pub fn display_weather_info(response: &WeatherResponse, format: &str){
5    // Extract the weather information from the response
6    let description: &String = &response.weather[0].description;
7    let temperature: f64 = response.main.temp;
8    let humidity: f64 = response.main.humidity;
9    let pressure: f64 = response.main.pressure;
10    let wind_speed: f64 = response.wind.speed;
11    // formatting weather information into a string
12    let weather_text: String = format!(
13        "Weather in {}: {} {}
14        > Temperature: {:.1}C,
15        > Humidity: {:.1}%,
16        > Pressure: {:.1}hPa, 
17        > Wind speed {:.1} m/s",
18        response.name,
19        description,
20        get_temp_emoji(temperature),
21        temperature, 
22        humidity,
23        pressure,
24        wind_speed,        
25    );
26    // Coloring the weather text based on weather conditions
27    let weather_text_colored = match description.as_str() {
28        "clear sky" => weather_text.bright_yellow(),
29        "few clouds" | "scattered clouds" | "broken clouds" => weather_text.bright_blue(),
30        "overcast clouds" | "mist" | "haze" | "smoke" | "sand" | "dust" | "fog" | "squalls" => weather_text.dimmed(),
31        "shower rain" | "rain" | "thunderstorm" | "snow" => weather_text.bright_cyan(),
32    _   => weather_text.normal(),
33    };
34    // Print the colored weather information 
35    // println!("{}", weather_text_colored);
36
37    match format {
38        "json" => {
39            let json: String = serde_json::to_string_pretty(&response).unwrap();
40            println!("{}", json);
41        },
42        "plain" => {
43            println!("{}", weather_text);
44        },
45        "human" => {
46            println!("{}", weather_text_colored);
47        }
48        _ => {
49            // Если формат неизвестен, выводим plain
50            println!("{}", weather_text);
51        }
52    }
53}

57fn get_temp_emoji(temperature: f64) -> &'static str {
58    if temperature < 0.0 {
59        "❄️"        // Снежинка
60    } else if temperature < 10.0 {
61        "🌥️"        // Облачно
62    } else if temperature < 20.0 {
63        "⛅"         // Солнце с облаком
64    } else if temperature < 30.0 {
65        "🌤️"        // Солнечно с лёгкой облачностью
66    } else {
67        "🔥"        // Жара
68    }
69}
